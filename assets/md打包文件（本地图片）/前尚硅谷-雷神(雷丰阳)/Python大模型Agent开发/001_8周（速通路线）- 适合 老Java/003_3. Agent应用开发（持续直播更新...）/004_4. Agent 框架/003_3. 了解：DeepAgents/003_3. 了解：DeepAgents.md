from deepagents import create_deep_agent
from langgraph.checkpoint.memory import MemorySaver
from deepagents.backends.filesystem import FilesystemBackend
from dotenv import load_dotenv
load_dotenv()
# Checkpointer is REQUIRED for human-in-the-loop
checkpointer = MemorySaver()


##virtual_mode 启用虚拟路径模式。
# 主要应用场景：与组合后端搭配使用时，提供稳定、不依赖后端的路径语义，
# 该模式会去除路由前缀，并将标准化路径转发至对应路由后端。
# 当设为True时，所有路径均视为以根目录为基准的虚拟路径。
# 禁止路径遍历（如使用..、~），且所有解析后的路径都会经过校验，确保不会超出根目录范围。
# 当设为False（默认值）时，绝对路径直接生效，相对路径在根目录下解析。
# 该模式无法阻止智能体访问根目录之外的路径，存在安全风险。
# 绝对路径（例如 /etc/passwd）会完全绕过根目录包含.. 的相对路径可脱离根目录限制。
# 智能体将拥有无限制的文件系统访问权限
agent = create_deep_agent(
    model="deepseek-chat",
    backend=FilesystemBackend(root_dir="C:\\Users\\53409\\PycharmProjects\\langchain-demo\\skills",
                              virtual_mode=True),
    skills=["C:\\Users\\53409\\PycharmProjects\\langchain-demo\\skills"],
    interrupt_on={
        "write_file": False,  # Default: approve, edit, reject
        "read_file": False,  # No interrupts needed
        "edit_file": False    # Default: approve, edit, reject
    },
    debug=True
    # checkpointer=checkpointer,  # Required!
)
