# 66、强化学习之GRPO算法

GRPO (Group Relative Policy Optimization) 是 DeepSeek-R1 论文中提出的一种新型强化学习算法。

核心思想是：不再需要传统的 Critic 网络（价值函数）来预测奖励，而是通过“群体对标”的方式，直接计算一组样本的相对优势。

```python
import random
import pygame

TRAP_COUNT = 3  # 陷阱数量
GRID_COUNT = 8  # 迷宫大小 (N x N)
SCREEN_WINDOW = 1000
CELL_SIZE = (SCREEN_WINDOW - 100) // GRID_COUNT
MAZE_SIZE = CELL_SIZE * GRID_COUNT
OFFSET_Y = 60  # 顶部文字区域高度

# 颜色定义
COLOR_BG = (245, 245, 245)
COLOR_TEXT = (44, 62, 80)
COLOR_LINE = (189, 195, 199)
COLOR_AGENT = (52, 152, 219)  # 蓝色
COLOR_GOAL = (46, 204, 113)  # 绿色
COLOR_TRAP = (231, 76, 60)  # 红色


## 初始化游戏界面
class MazeGUI:
    def __init__(self, traps, goal):
        pygame.init()
        self.screen = pygame.display.set_mode((MAZE_SIZE, MAZE_SIZE + OFFSET_Y))
        pygame.display.set_caption("秘境寻宝")
        self.font = pygame.font.SysFont("SimHei", 20)
        self.clock = pygame.time.Clock()
        self.traps = traps
        self.goal = goal

    def draw(self, position, episode, step):
        self.screen.fill(COLOR_BG)

        # 1. 状态信息
        info = self.font.render(f"episode: {episode + 1} | step: {step} | trap: {len(self.traps)}", True, COLOR_TEXT)
        self.screen.blit(info, (10, 15))

        # 2. 网格和元素
        for r in range(GRID_COUNT):
            for c in range(GRID_COUNT):
                rect = (c * CELL_SIZE, r * CELL_SIZE + OFFSET_Y, CELL_SIZE, CELL_SIZE)
                pygame.draw.rect(self.screen, COLOR_LINE, rect, 1)  # 画边框

                # 画陷阱
                if (r, c) in self.traps:
                    pygame.draw.rect(self.screen, COLOR_TRAP, (rect[0] + 4, rect[1] + 4, CELL_SIZE - 8, CELL_SIZE - 8))
                # 画终点
                elif (r, c) == self.goal:
                    pygame.draw.rect(self.screen, COLOR_GOAL, (rect[0] + 4, rect[1] + 4, CELL_SIZE - 8, CELL_SIZE - 8))

        # 3. 画探险家
        if position != 'terminal':
            r, c = eval(position)
            center = (c * CELL_SIZE + CELL_SIZE // 2, r * CELL_SIZE + OFFSET_Y + CELL_SIZE // 2)
            pygame.draw.circle(self.screen, COLOR_AGENT, center, CELL_SIZE // 3)

        pygame.display.flip()

def init_environment():
    """初始化陷阱和终点坐标"""
    all_positions = [(r, c) for r in range(GRID_COUNT) for c in range(GRID_COUNT)]
    all_positions.remove((0, 0))  # 起点不能是陷阱
    goal = (GRID_COUNT - 1, GRID_COUNT - 1)
    all_positions.remove(goal)  # 终点不能是陷阱

    # 随机选出陷阱
    traps = random.sample(all_positions, min(TRAP_COUNT, len(all_positions)))
    # traps = [(2, 1), (3,4), (1,6)]
    return traps, goal

# 与环境交互，获取下一个状态和奖励
def get_env_feedback(current_state, action, goal, traps):
    # current_state是字符串"(0, 0)"，用eval函数解析，r_idx表示当前状态的行索引，c_idx表示当前状态的列索引
    r_idx, c_idx = eval(current_state)

    # 根据动作决定下一个位置的坐标
    if action == 'up':
        r_idx = max(0, r_idx - 1)
    elif action == 'down':
        r_idx = min(GRID_COUNT - 1, r_idx + 1)
    elif action == 'left':
        c_idx = max(0, c_idx - 1)
    elif action == 'right':
        c_idx = min(GRID_COUNT - 1, c_idx + 1)

    next_state = (r_idx, c_idx)

    is_terminated = False
    if next_state == goal:
        # 得到宝藏
        reward = 1
        is_terminated = True
    elif next_state in traps:
        # 遇到陷阱
        reward = -1
        is_terminated = True
    else:
        # 普通步骤
        # reward = 0
        # reward = -0.01  # 扣太少，可能导致原地踏步
        reward = -0.1

    return str(next_state), reward, is_terminated
```

```python
import numpy as np
from torch.distributions import Categorical
from torch import optim
from torch import nn
import torch

ACTIONS = ['up', 'down', 'left', 'right']
traps, goal = init_environment()
gui = MazeGUI(traps, goal)
GAMMA = 0.99
EPS = 0.2        # PPO 裁剪阈值
K_EPOCHS = 10    # 每条轨迹重复学习多少次

# --- 修改后的网络：去掉 Critic 部分 ---
class ActorNet(nn.Module):
    def __init__(self, n_states, n_actions):
        super(ActorNet, self).__init__()
        self.base = nn.Sequential(
            nn.Linear(n_states, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU()
        )
        self.actor = nn.Sequential(
            nn.Linear(128, n_actions),
            nn.Softmax(dim=-1)
        )
        # 注意：这里删除了 self.critic

    def forward(self, x):
        features = self.base(x)
        probs = self.actor(features)
        return probs # 只返回概率

def state_to_tensor(state_str):
    r, c = eval(state_str)
    one_hot = torch.zeros(GRID_COUNT * GRID_COUNT)
    one_hot[r * GRID_COUNT + c] = 1.0
    return one_hot

# --- 配置 GRPO 参数 ---
GROUP_SIZE = 8  # 每次采样多少个“同伴”进行对比
model = ActorNet(GRID_COUNT * GRID_COUNT, len(ACTIONS))
optimizer = optim.Adam(model.parameters(), lr=0.0005)
success_count = 0

# --- 训练循环 ---
for episode in range(1000):
    group_rewards = []
    group_data = [] # 存储这一组内所有轨迹的数据

    # 群体采样：针对同一个起点，采样 G 条轨迹
    for g in range(GROUP_SIZE):
        current_state = "(0, 0)"
        states = []
        actions = []
        old_log_probs = []
        rewards = []

        for t in range(50):
            state_tensor = state_to_tensor(current_state)
            with torch.no_grad(): # 采样阶段不计梯度
                probs = model(state_tensor)
                m = Categorical(probs)
                action = m.sample()

            next_state, reward, is_terminated = get_env_feedback(current_state, ACTIONS[action.item()], goal, traps)

            states.append(state_tensor)
            actions.append(action)
            old_log_probs.append(m.log_prob(action))
            rewards.append(reward)

            current_state = next_state
            if is_terminated:
                if reward > 0: success_count += 1
                break

        # 计算该轨迹的总回报，没有计算该轨迹中每个步骤的回报了
        G_return = 0
        for r in reversed(rewards): G_return = r + GAMMA * G_return

        # 将该轨迹或的总回报、状态、动作、概率分布都记录下来
        group_rewards.append(G_return)
        group_data.append((states, actions, old_log_probs))

    # GRPO 核心：计算群体相对优势 (Relative Advantage)
    # 计算这一组回报的均值和标准差
    mean_reward = np.mean(group_rewards)
    std_reward = np.std(group_rewards) + 1e-8

    # 3. 更新模型
    for _ in range(K_EPOCHS):
        total_loss = 0

        # 遍历组里面的每条轨迹
        for i in range(GROUP_SIZE):
            # 获取该轨迹的 Advantage：(当前回报 - 组平均) / 组标准差
            # 也就是得到了该条轨迹的相对优势，然后就可以利用该相对优势来更新模型了（后面流程和PPO一样）
            advantage = (group_rewards[i] - mean_reward) / std_reward

            states_b, actions_b, old_log_probs_b = group_data[i]
            batch_states = torch.stack(states_b)
            batch_actions = torch.stack(actions_b)
            batch_old_log_probs = torch.stack(old_log_probs_b)

            new_probs = model(batch_states)
            new_m = Categorical(new_probs)
            new_log_probs = new_m.log_prob(batch_actions)
            ratio = torch.exp(new_log_probs - batch_old_log_probs)
            surr1 = ratio * advantage
            surr2 = torch.clamp(ratio, 1 - EPS, 1 + EPS) * advantage

            # GRPO 通常还会加入 KL 散度约束，这里简化为 PPO 的剪切 Loss
            actor_loss = -torch.min(surr1, surr2).mean()
            total_loss += actor_loss

        optimizer.zero_grad()
        (total_loss / GROUP_SIZE).backward() # 对组内 Loss 取平均
        optimizer.step()

    if episode % 100 == 0:
        print(f"Episode {episode}, Successes: {success_count}")
```

```python
current_state = "(0, 0)"
is_terminated = False

step = 0
while not is_terminated:
    gui.draw(current_state, 1, step)
    gui.clock.tick(5)  # 推理时速度放慢，方便观察 (5 FPS)

    with torch.no_grad():
        state_tensor = state_to_tensor(current_state)
        probs = model(state_tensor)
        action_idx = torch.argmax(probs).item()
        action = ACTIONS[action_idx]

    next_state, reward, is_terminated = get_env_feedback(current_state, action, goal, traps)
    current_state = next_state
    step += 1

    if is_terminated:
        gui.draw(current_state, 1, step)  # 画出最后一步
        pygame.time.delay(1000)  # 终点停留 1 秒
        if reward == 1:
            print(f"成功到达终点！步数：{step}")
        else:
            print(f"不幸掉入陷阱！")
```
