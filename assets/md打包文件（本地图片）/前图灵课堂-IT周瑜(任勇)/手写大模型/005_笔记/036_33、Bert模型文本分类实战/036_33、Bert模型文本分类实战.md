```python
class BertDataset(Dataset):
    def __init__(self, dataset):
        self.dataset = dataset

    def __len__(self):
        return len(self.dataset)

    def __getitem__(self, idx):
        item = self.dataset[idx]
        text = item['sentence']
        # 使用BERT分词器编码文本
        encoding = tokenizer.encode_plus(
            text,
            add_special_tokens=True,  # 添加[CLS]和[SEP]
            max_length=128,  # 最大长度
            padding='max_length',  # 填充到最大长度
            truncation=True,  # 截断过长的文本
            return_attention_mask=True,  # 返回注意力掩码
            return_tensors='pt'  # 返回PyTorch张量
        )

        return encoding['input_ids'][0], encoding['attention_mask'][0], torch.tensor(item['label'])


train_dataset = BertDataset(ds_train_torch)
train_loader = DataLoader(train_dataset, batch_size=2, shuffle=False)
```

## 定义模型

```python
# 定义模型
from torch import nn

from modelscope import AutoModel

class BertForTextClassification(nn.Module):
    def __init__(self):
        super().__init__()
        self.bert = AutoModel.from_pretrained("tiansz/bert-base-chinese")
        self.fc = nn.Linear(768, 2)

    def forward(self, input_ids, attention_mask):
        outputs = self.bert(input_ids, attention_mask=attention_mask)
        pooled_output = outputs.pooler_output
        return self.fc(pooled_output)
```

## 开始训练

```python
device = torch.device('cuda' if torch.cuda.is_available() else 'mps')

model = BertForTextClassification()
model.to(device)

optimizer = torch.optim.Adam(model.parameters(), lr=2e-5)
criterion = nn.CrossEntropyLoss()

EPOCHS = 10
for epoch in range(EPOCHS):
    for i, (input_ids, attention_mask, labels) in enumerate(train_loader):
        input_ids = input_ids.to(device)
        attention_mask = attention_mask.to(device)
        labels = labels.to(device)

        outputs = model(input_ids, attention_mask)
        loss = criterion(outputs, labels.long())
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        if (i + 1) % 10 == 0:
            print('Epoch [{}/{}], Step [{}/{}], Loss: {:.4f}'
                  .format(epoch + 1, EPOCHS, i + 1, len(train_loader), loss.item()))
```
