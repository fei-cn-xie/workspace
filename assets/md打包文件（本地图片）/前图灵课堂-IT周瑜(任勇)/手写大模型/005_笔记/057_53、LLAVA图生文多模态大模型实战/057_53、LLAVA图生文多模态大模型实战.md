```python
from torch.utils.data import Dataset
from PIL import Image

class LLaVADataset(Dataset):
    def __init__(self, data, clip_processor, qwen_tokenizer):
        self.data = data
        self.clip_processor = clip_processor
        self.qwen_tokenizer = qwen_tokenizer

    def __len__(self):
        return len(self.data)

    def __getitem__(self, idx):
        image_path, caption = self.data[idx]

        image = Image.open(image_path)
        pixel_values = self.clip_processor(images=image, return_tensors="pt")['pixel_values']

        text_tokenized = self.qwen_tokenizer(caption, return_tensors="pt", padding=True, truncation=True,
                                             max_length=128)

        return pixel_values.squeeze(0), text_tokenized['input_ids'].squeeze(0), text_tokenized[
            'attention_mask'].squeeze(0)
```

```python
import torch
from torch import nn

# 大都督周瑜（我的微信: it_zhouyu）

class ZhouyuLLaVAModel(nn.Module):
    def __init__(self, clip_model, qwen_model, qwen_tokenizer):
        super().__init__()
        self.image_model = clip_model.vision_model
        self.language_model = qwen_model
        self.qwen_tokenizer = qwen_tokenizer

        # 线性投影层，用于对齐维度
        self.image_proj = nn.Linear(
            self.image_model.config.hidden_size,
            self.language_model.config.hidden_size
        )

        # 获取特殊 token ID
        self.img_token_id = self.qwen_tokenizer.encode("<|IMG|>")[0]

        # 冻结大部分参数，只训练投影层
        self.freeze_models()

    def freeze_models(self):
        # 冻结视觉模型
        for param in self.image_model.parameters():
            param.requires_grad = False

        # 冻结语言模型
        for param in self.language_model.parameters():
            param.requires_grad = False
        # # 冻结语言模型的部分层
        # for name, param in self.language_model.named_parameters():
        #     if "layer.11" in name:
        #         param.requires_grad = False


        # 投影层是需要训练的
        for param in self.image_proj.parameters():
            param.requires_grad = True

    def forward(self, pixel_values, text_input_ids, attention_mask):
        # 1. 获取图像特征
        image_features = self.image_model(pixel_values=pixel_values, return_dict=True).pooler_output

        # 2. 投影图像特征
        projected_image_features = self.image_proj(image_features)

        # 3. 获取文本特征
        text_embeddings = self.language_model.get_input_embeddings()(text_input_ids)

        # 4. 融合图像与文本特征
        final_embeddings = []
        for i in range(text_embeddings.shape[0]):  # 遍历 batch
            # 找到 <|IMG|> token 的位置
            img_token_pos = (text_input_ids[i] == self.img_token_id).nonzero(as_tuple=True)[0][0]

            # 替换 <|IMG|> 的 embedding 为投影后的图片特征
            current_embedding = torch.cat([
                text_embeddings[i, :img_token_pos],
                projected_image_features[i].unsqueeze(0),
                text_embeddings[i, img_token_pos + 1:]
            ], dim=0)
            final_embeddings.append(current_embedding)
        inputs_embeds = torch.stack(final_embeddings)

        # 5. 准备Labels
        labels = text_input_ids.clone()
        for i in range(labels.shape[0]):
            img_token_pos = (labels[i] == self.img_token_id).nonzero(as_tuple=True)[0]
            pos = img_token_pos[0]
            # 屏蔽掉问题部分，只在答案部分计算 Loss
            labels[i, :pos + 1] = -100

        # 6. 将融合后的 embedding 输入 LLM
        outputs = self.language_model(
            inputs_embeds=inputs_embeds,
            attention_mask=attention_mask,
            labels=labels,
            return_dict=True
        )

        return outputs.loss
```

```python
def generate(model, pixel_values, prompt_text):

    prompt_ids = model.qwen_tokenizer(prompt_text, return_tensors="pt")['input_ids']


    image_features = model.image_model(pixel_values=pixel_values, return_dict=True).pooler_output
    projected_image_features = model.image_proj(image_features)


    prompt_embeddings = model.language_model.get_input_embeddings()(prompt_ids)


    img_token_pos = (prompt_ids[0] == model.img_token_id).nonzero(as_tuple=True)[0][0]
    inputs_embeds = torch.cat([
        prompt_embeddings[0, :img_token_pos],
        projected_image_features[0].unsqueeze(0),
        prompt_embeddings[0, img_token_pos + 1:]
    ], dim=0).unsqueeze(0)

    with torch.no_grad():
        outputs = model.language_model.generate(
            inputs_embeds=inputs_embeds,
            max_new_tokens=50,
            pad_token_id=model.qwen_tokenizer.pad_token_id,
            eos_token_id=model.qwen_tokenizer.eos_token_id
        )

    return model.qwen_tokenizer.decode(outputs[0], skip_special_tokens=True)
```
