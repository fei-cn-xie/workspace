```python
from torch import nn

class TextEncoder(nn.Module):
    def __init__(self, vocab_size, embed_dim=64, max_len=5, nhead=2, num_layers=1):
        super().__init__()
        self.embed_dim = embed_dim
        self.token_embedding = nn.Embedding(vocab_size, embed_dim)
        self.position_embedding = nn.Embedding(max_len, embed_dim)

        encoder_layer = nn.TransformerEncoderLayer(
            d_model=embed_dim,
            nhead=nhead,
            dim_feedforward=2048,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers)

    def forward(self, x):
        batch_size, seq_len = x.shape

        token_embeddings = self.token_embedding(x)

        pos_ids = torch.arange(0, seq_len).unsqueeze(0).repeat(batch_size, 1)
        pos_embeddings = self.position_embedding(pos_ids)

        embedding = token_embeddings + pos_embeddings
        x = self.transformer(embedding)
        x = x.mean(dim=1)
        return x
```

class ImageEncoder(nn.Module):
    def __init__(self, image_size=28, patch_size=7, in_channels=1, num_classes=10,
                 embed_dim=64, num_layers=4, num_heads=8, dim_feedforward=2048):
        super().__init__()
        assert image_size % patch_size == 0, "Image size must be divisible by patch size"

        self.image_size = image_size
        self.patch_size = patch_size
        self.embed_dim = embed_dim
        self.num_patches = (image_size // patch_size) ** 2

        # 通过卷积层切分为16张图片
        self.patch_conv = nn.Conv2d(in_channels, self.num_patches, kernel_size=4, stride=4)
        self.patch_embed = nn.Linear(in_channels * patch_size * patch_size, embed_dim)
        self.cls_token = torch.zeros(1, 1, embed_dim)
        self.pos_embed = nn.Embedding(self.num_patches + 1, embed_dim)

        # TransformerEncoder
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=embed_dim,
            nhead=num_heads,
            dim_feedforward=dim_feedforward,
            batch_first=True
        )
        self.encoder = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)

        # 分类头，不需要分类头了
        # self.head = nn.Linear(embed_dim, num_classes)

    def forward(self, x):
        batch_size = x.shape[0]  # batch size

        # 利用卷积层切分为16张图片
        x = self.patch_conv(x)  # [batch_size, 1, 28, 28] -> [batch_size, 16, 7, 7]
        x = x.view(batch_size, self.num_patches, -1) # [batch_size, 16, 7, 7] -> [batch_size, 16, 49]

        # 将图片转成向量
        x = self.patch_embed(x)  # [batch_size, 16, 49] -> [batch_size, 16, 64]

        # 加入cls_tokens
        cls_tokens = self.cls_token.expand(batch_size, -1, -1)  # [1, 1, embed_dim] -> [batch_size, 1, embed_dim]
        x = torch.cat((cls_tokens, x), dim=1)  # [batch_size, 16, 64] -> [batch_size, 17, 64]

        # 加上位置编码
        seq_len = x.shape[1]
        pos_ids = torch.arange(0, seq_len).unsqueeze(0).repeat(batch_size, 1)
        pos_embeddings = self.pos_embed(pos_ids)
        x = x + pos_embeddings

        # 送给TransformerEncoder进行编码，得到新的向量
        x = self.encoder(x)  # [batch_size, 17, 64]

        # 取出cls_tokens对应的向量
        cls_output = x[:, 0]

        # 进行分类，不需要分类头了
        # logits = self.head(cls_output)

        return cls_output

```python
class ZhouyuCLIP(nn.Module):
    def __init__(self, vocab_size):
        super().__init__()
        self.image_encoder = ImageEncoder(embed_dim=64)
        self.text_encoder = TextEncoder(vocab_size, embed_dim=64)

    def forward(self, texts, images):
        text_features = self.text_encoder(texts)
        image_features = self.image_encoder(images)
        return text_features, image_features
```

```python
# 对比学习
def contrastive_loss(text_features, image_features):
    logits = image_features @ text_features.T
    labels = torch.arange(logits.size(0))
    loss_i2t = nn.CrossEntropyLoss()(logits, labels)
    loss_t2i = nn.CrossEntropyLoss()(logits.T, labels)
    return (loss_i2t + loss_t2i) / 2
```

## 对比学习

假如有两张图片img1, img2
假如有两段文本text1, text2
我们期望模型能学会img1和text1相似，img2和text2相似

logits如下
img1*text1，img1*text2
img2*text1，img2*text2

logits.T如下
img1*text1，img2*text1
img1*text2，img2*text2

## 为什么要算两次交叉熵

loss_i2t的作用是让ImageEncoder更聪明
loss_t2i的作用是让TextEncoder更聪明

```python
model = ZhouyuCLIP(vocab_size=len(vocab))
optimizer = torch.optim.Adam(model.parameters(), lr=3e-4)

for epoch in range(100):
    total_loss = 0
    for text_batch, img_batch in dataloader:

        optimizer.zero_grad()
        text_features, image_features = model(text_batch, img_batch)
        loss = contrastive_loss(text_features, image_features)

        loss.backward()
        optimizer.step()

        total_loss += loss.item()

    if epoch % 10 == 0:
        print(f"Epoch {epoch}, Loss: {total_loss/len(dataloader):.4f}")
```
