## VAE

VAE（Variational Autoencoder，变分自编码器）是一种生成式深度学习模型，属于无监督学习方法，可用于图片生成。

```python
import torch
import torch.nn as nn

class VAE(nn.Module):
    def __init__(self, in_channels=1, latent_channels=4):
        super().__init__()
        self.latent_channels = latent_channels
        # Encoder
        self.encoder = nn.Sequential(
            nn.Conv2d(in_channels, 32, kernel_size=4, stride=2, padding=1),  # 32x32 -> 16x16
            nn.ReLU(),
            nn.Conv2d(32, 64, kernel_size=4, stride=2, padding=1),  # 16x16 -> 8x8
            nn.ReLU(),
            nn.Conv2d(64, 128, kernel_size=4, stride=2, padding=1),  # 8x8 -> 4x4
            nn.ReLU(),
            nn.Flatten(),
            nn.Linear(128 * 4 * 4, 256)
        )
        self.fc_mu = nn.Linear(256, latent_channels * 4 * 4)
        self.fc_log_var = nn.Linear(256, latent_channels * 4 * 4)

        # Decoder
        self.decoder_input = nn.Linear(latent_channels * 4 * 4, 128 * 4 * 4)
        self.decoder = nn.Sequential(
            nn.Unflatten(1, (128, 4, 4)),
            nn.ConvTranspose2d(128, 64, kernel_size=4, stride=2, padding=1),  # 4x4 -> 8x8
            nn.ReLU(),
            nn.ConvTranspose2d(64, 32, kernel_size=4, stride=2, padding=1),  # 8x8 -> 16x16
            nn.ReLU(),
            nn.ConvTranspose2d(32, in_channels, kernel_size=4, stride=2, padding=1),  # 16x16 -> 32x32
            nn.Tanh()  # Output values between -1 and 1
        )

    def encode(self, x):
        h = self.encoder(x)
        mu = self.fc_mu(h)
        log_var = self.fc_log_var(h)
        return mu, log_var

    def reparameterize(self, mu, log_var):
        std = torch.exp(0.5 * log_var)
        eps = torch.randn_like(std)
        return mu + eps * std

    def decode(self, z):
        h = self.decoder_input(z)
        return self.decoder(h)

    def forward(self, x):
        mu, log_var = self.encode(x)
        z = self.reparameterize(mu, log_var)
        return self.decode(z), mu, log_var

    def get_latent(self, x):
        mu, log_var = self.encode(x)
        z = self.reparameterize(mu, log_var)
        return z.view(-1, self.latent_channels, 4, 4)
```

```python
from torchvision import datasets, transforms
from torch.utils.data import DataLoader

transform = transforms.Compose([
    transforms.Resize(32),
    transforms.ToTensor(),
    transforms.Normalize([0.5], [0.5])  # Normalize to [-1, 1]
])
dataset = datasets.MNIST(root='./data', train=True, download=True, transform=transform)
vae_dataloader = DataLoader(dataset, batch_size=256, shuffle=True, num_workers=4)
```

```python
from torchvision.utils import save_image
import os
import torch.nn.functional as F

IN_CHANNELS = 1
LATENT_CHANNELS = 4
# VAE_EPOCHS = 25
VAE_EPOCHS = 1

os.makedirs("vae_samples", exist_ok=True)
os.makedirs("final_weights", exist_ok=True)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

vae = VAE(in_channels=IN_CHANNELS, latent_channels=LATENT_CHANNELS).to(device)
vae_optimizer = torch.optim.Adam(vae.parameters(), lr=1e-3)

def vae_loss_function(recon_x, x, mu, log_var):
    recon_loss = F.mse_loss(recon_x, x, reduction='sum')
    kld = -0.5 * torch.sum(1 + log_var - mu.pow(2) - log_var.exp())
    return recon_loss + kld


for epoch in range(VAE_EPOCHS):
    vae.train()
    for images, _ in vae_dataloader:
        images = images.to(device)
        vae_optimizer.zero_grad()
        recon_images, mu, log_var = vae(images)
        loss = vae_loss_function(recon_images, images, mu, log_var)
        loss.backward()
        vae_optimizer.step()

    if (epoch + 1) % 5 == 0:
        vae.eval()
        with torch.no_grad():
            fixed_images, _ = next(iter(vae_dataloader))
            recon_fixed, _, _ = vae(fixed_images.to(device))
            comparison = torch.cat([fixed_images[:8], recon_fixed.cpu()[:8]])
            save_image(comparison, f"vae_samples/epoch_{epoch + 1}.png", nrow=8, normalize=True)

vae_path = "final_weights/mnist_vae.pth"
torch.save(vae.state_dict(), vae_path)
print(f"VAE模型训练完成，模型保存在了{vae_path}")
```
