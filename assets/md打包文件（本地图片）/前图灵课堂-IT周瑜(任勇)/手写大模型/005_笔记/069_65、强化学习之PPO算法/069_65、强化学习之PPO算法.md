# 65、强化学习之PPO算法

## PPO

Proximal Policy Optimization, 近端策略优化。

```python
import random
import pygame

TRAP_COUNT = 3  # 陷阱数量
GRID_COUNT = 8  # 迷宫大小 (N x N)
SCREEN_WINDOW = 1000
CELL_SIZE = (SCREEN_WINDOW - 100) // GRID_COUNT
MAZE_SIZE = CELL_SIZE * GRID_COUNT
OFFSET_Y = 60  # 顶部文字区域高度

# 颜色定义
COLOR_BG = (245, 245, 245)
COLOR_TEXT = (44, 62, 80)
COLOR_LINE = (189, 195, 199)
COLOR_AGENT = (52, 152, 219)  # 蓝色
COLOR_GOAL = (46, 204, 113)  # 绿色
COLOR_TRAP = (231, 76, 60)  # 红色


## 初始化游戏界面
class MazeGUI:
    def __init__(self, traps, goal):
        pygame.init()
        self.screen = pygame.display.set_mode((MAZE_SIZE, MAZE_SIZE + OFFSET_Y))
        pygame.display.set_caption("秘境寻宝")
        self.font = pygame.font.SysFont("SimHei", 20)
        self.clock = pygame.time.Clock()
        self.traps = traps
        self.goal = goal

    def draw(self, position, episode, step):
        self.screen.fill(COLOR_BG)

        # 1. 状态信息
        info = self.font.render(f"episode: {episode + 1} | step: {step} | trap: {len(self.traps)}", True, COLOR_TEXT)
        self.screen.blit(info, (10, 15))

        # 2. 网格和元素
        for r in range(GRID_COUNT):
            for c in range(GRID_COUNT):
                rect = (c * CELL_SIZE, r * CELL_SIZE + OFFSET_Y, CELL_SIZE, CELL_SIZE)
                pygame.draw.rect(self.screen, COLOR_LINE, rect, 1)  # 画边框

                # 画陷阱
                if (r, c) in self.traps:
                    pygame.draw.rect(self.screen, COLOR_TRAP, (rect[0] + 4, rect[1] + 4, CELL_SIZE - 8, CELL_SIZE - 8))
                # 画终点
                elif (r, c) == self.goal:
                    pygame.draw.rect(self.screen, COLOR_GOAL, (rect[0] + 4, rect[1] + 4, CELL_SIZE - 8, CELL_SIZE - 8))

        # 3. 画探险家
        if position != 'terminal':
            r, c = eval(position)
            center = (c * CELL_SIZE + CELL_SIZE // 2, r * CELL_SIZE + OFFSET_Y + CELL_SIZE // 2)
            pygame.draw.circle(self.screen, COLOR_AGENT, center, CELL_SIZE // 3)

        pygame.display.flip()

def init_environment():
    """初始化陷阱和终点坐标"""
    all_positions = [(r, c) for r in range(GRID_COUNT) for c in range(GRID_COUNT)]
    all_positions.remove((0, 0))  # 起点不能是陷阱
    goal = (GRID_COUNT - 1, GRID_COUNT - 1)
    all_positions.remove(goal)  # 终点不能是陷阱

    # 随机选出陷阱
    traps = random.sample(all_positions, min(TRAP_COUNT, len(all_positions)))
    # traps = [(2, 1), (3,4), (1,6)]
    return traps, goal

# 与环境交互，获取下一个状态和奖励
def get_env_feedback(current_state, action, goal, traps):
    # current_state是字符串"(0, 0)"，用eval函数解析，r_idx表示当前状态的行索引，c_idx表示当前状态的列索引
    r_idx, c_idx = eval(current_state)

    # 根据动作决定下一个位置的坐标
    if action == 'up':
        r_idx = max(0, r_idx - 1)
    elif action == 'down':
        r_idx = min(GRID_COUNT - 1, r_idx + 1)
    elif action == 'left':
        c_idx = max(0, c_idx - 1)
    elif action == 'right':
        c_idx = min(GRID_COUNT - 1, c_idx + 1)

    next_state = (r_idx, c_idx)

    is_terminated = False
    if next_state == goal:
        # 得到宝藏
        reward = 1
        is_terminated = True
    elif next_state in traps:
        # 遇到陷阱
        reward = -1
        is_terminated = True
    else:
        # 普通步骤
        # reward = 0
        # reward = -0.01  # 扣太少，可能导致原地踏步
        reward = -0.1

    return str(next_state), reward, is_terminated
```

```python
from torch.distributions import Categorical
from torch import optim
from torch import nn
import torch

ACTIONS = ['up', 'down', 'left', 'right']
traps, goal = init_environment()
gui = MazeGUI(traps, goal)
GAMMA = 0.99
EPS = 0.2        # PPO 裁剪阈值
K_EPOCHS = 10    # 每条轨迹重复学习多少次

class ActorCriticNet(nn.Module):
    def __init__(self, n_states, n_actions):
        super(ActorCriticNet, self).__init__()
        self.base = nn.Sequential(
            nn.Linear(n_states, 128),
            nn.ReLU(),
            nn.Linear(128, 128),
            nn.ReLU()
        )
        self.actor = nn.Sequential(
            nn.Linear(128, n_actions),
            nn.Softmax(dim=-1)
        )
        self.critic = nn.Sequential(
            nn.Linear(128, 1)
        )

    def forward(self, x):
        features = self.base(x)
        probs = self.actor(features)
        value = self.critic(features)
        return probs, value

def state_to_tensor(state_str):
    r, c = eval(state_str)
    one_hot = torch.zeros(GRID_COUNT * GRID_COUNT)
    one_hot[r * GRID_COUNT + c] = 1.0
    return one_hot

model = ActorCriticNet(GRID_COUNT * GRID_COUNT, len(ACTIONS))
optimizer = optim.Adam(model.parameters(), lr=0.0005) # PPO通常学习率可以设稍微小一点更稳
success_count = 0

for episode in range(3000):
    current_state = "(0, 0)"
    step = 0

    states = []
    actions = []
    old_log_probs = []
    values = []
    rewards = []

    for t in range(20):
        gui.draw(current_state, episode, step) # 训练快可以关掉渲染
        state_tensor = state_to_tensor(current_state)
        probs, value = model(state_tensor)

        m = Categorical(probs)
        action = m.sample()

        next_state, reward, is_terminated = get_env_feedback(current_state, ACTIONS[action.item()], goal, traps)

        # 存储当前步的数据
        states.append(state_tensor)
        actions.append(action)
        old_log_probs.append(m.log_prob(action).detach()) # 必须detach，因为它是作为参考的基准
        values.append(value)
        rewards.append(reward)

        current_state = next_state
        step += 1
        if is_terminated:
            if reward > 0: success_count += 1
            break

    # 计算这一个回合中每个步骤的奖励
    returns = []
    G = 0
    for r in reversed(rewards):
        G = r + GAMMA * G
        returns.insert(0, G)

    # 转换为 Tensor 方便计算
    batch_states = torch.stack(states)
    batch_actions = torch.stack(actions).unsqueeze(1)
    batch_returns = torch.tensor(returns).unsqueeze(1)
    batch_values = torch.stack(values)
    batch_old_log_probs = torch.stack(old_log_probs).unsqueeze(1)

    # --- 循环外计算Advantage ---
    advantage = batch_returns - batch_values.detach()

    # PPO 核心：多次更新（K_EPOCHS）
    # A2C算法只会利用advantage更新一次，而PPO算法会更新K_EPOCHS次
    # 第一次循环，ratio为1，利用原始advantage更新一次模型
    # 第二次循环，ratio就可能不为1了，但本质上还是在利用advantage更新模型，只不过更新幅度会限制住
    # PPO算法的思想是：如果跑出来的轨迹效果好，那就多提升几次动作概率，如果跑出来的轨迹效果差，那就多减少几次动作概率。
    for _ in range(K_EPOCHS):
        # 重新把 batch_states 丢进模型，得到“新”的概率和价值
        new_probs, new_values = model(batch_states)

        # 计算 Ratio (新动作概率 / 旧动作概率)
        # loga - logb = log(a/b)
        new_m = Categorical(new_probs)
        new_log_probs = new_m.log_prob(batch_actions.squeeze()).unsqueeze(1)
        ratio = torch.exp(new_log_probs - batch_old_log_probs)

        # 计算 PPO 的两个 Loss 部分
        surr1 = ratio * advantage
        surr2 = torch.clamp(ratio, 1 - EPS, 1 + EPS) * advantage

        # Actor Loss: 取两者的最小值，防止更新过猛
        # 如果 ratio >= 1.2， actor_loss都等于1.2*advantage，模型就会“停止更新”，actor_loss不再发生变化就表示模型停止更新了，尽管actor_loss有值
        # 也就说PPO算法中，一条轨迹会利用多次，但是动作概率顶多提升20%，相比A2C算法，A2C算法只会利用一次轨迹，概率也只提升一次（可能少了，可能多了）
        actor_loss = -torch.min(surr1, surr2).mean()

        # Critic Loss: 预测的价值要尽可能接近实际 Returns
        critic_loss = nn.functional.mse_loss(new_values, batch_returns)

        # 总 Loss
        loss = actor_loss + critic_loss

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

    if episode % 100 == 0:
        print(f"Episode {episode}, Successes: {success_count}")

print("训练完成！")
```

我说说我对PPO的理解，

PPO的工作流程是：

1. 先跑一次实际的轨迹出来，并且记录这条轨迹执行的动作、状态、奖励、和动作的概率分布
2. 计算这条实际轨迹相对当前模型的优势，看是比之前跑得好，还是跑得差
3. 假如是跑得好，那就要更新模型参数，让模型学会跑得更好
4. 而模型要学会，就是要提升刚刚这条轨迹中执行的动作的概率
5. PPO算法没有一次性将概率提升，而是分成了多次
6. 第一次，ratio=1，会直接用计算出来的advantages来提升动作概率（应该提升为了实际轨迹中发生的动作概率）
7. 第二次，ratio应该就会大于1了，因为第一次提升了动作概率，ratio如果大于1，如果特别大，其实也会控制在1 + EPS，因此ratio不会特别大，然后再次提升概率
8. 也就是说，如果我实际跑出来一条轨迹，发现比之前跑得好，我会提升很多次这条轨迹中动作发生的概率，而不是像A2C算法那样只提升了一次
9. 相反，如果实际轨迹比之前跑得差，那就要降低动作概率
10. 第一次，ratio=1，会直接用计算出来的advantages来降低动作概率
11. 第二次，ratio应该就会小于1，然后就继续降低动作概率
