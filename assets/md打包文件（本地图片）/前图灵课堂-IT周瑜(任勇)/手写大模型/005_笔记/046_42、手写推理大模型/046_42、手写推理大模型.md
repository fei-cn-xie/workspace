```python
zhouyu_chat_template = """<|zhouyu_start|>user
{question}<|zhouyu_end|>
<|zhouyu_start|>assistant
<think>
{think}
</think>
{answer}
<|zhouyu_end|>"""
```

```python
from torch.utils.data import Dataset, DataLoader

# 微信id：it_zhouyu
class ZhouyuDataset(Dataset):
    def __init__(self, data, max_length=256):
        self.encodings = []
        for qa in data:
            text = zhouyu_chat_template.format(question=qa["question"], think=qa["think"], answer=qa["answer"])
            encoded = tokenizer(
                text,
                max_length=max_length,
                padding='max_length',
                truncation=True,
                return_tensors='pt'
            )
            input_ids = encoded['input_ids'].squeeze()
            self.encodings.append(input_ids)

    def __len__(self):
        return len(self.encodings)

    def __getitem__(self, idx):
        return self.encodings[idx]


dataset = ZhouyuDataset(data)
```

```python
generated_ids = model.generate(**model_inputs, max_new_tokens=200, pad_token_id=tokenizer.pad_token_id, eos_token_id=tokenizer.eos_token_id)
```
