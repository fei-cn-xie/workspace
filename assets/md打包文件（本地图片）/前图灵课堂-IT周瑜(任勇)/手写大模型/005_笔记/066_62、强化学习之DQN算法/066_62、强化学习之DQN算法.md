```python
import pygame

GRID_COUNT = 8  # 迷宫大小 (N x N)
SCREEN_WINDOW = 1000
CELL_SIZE = (SCREEN_WINDOW - 100) // GRID_COUNT
MAZE_SIZE = CELL_SIZE * GRID_COUNT
OFFSET_Y = 60  # 顶部文字区域高度

# 颜色定义
COLOR_BG = (245, 245, 245)
COLOR_TEXT = (44, 62, 80)
COLOR_LINE = (189, 195, 199)
COLOR_AGENT = (52, 152, 219)  # 蓝色
COLOR_GOAL = (46, 204, 113)  # 绿色
COLOR_TRAP = (231, 76, 60)  # 红色


## 初始化游戏界面
class MazeGUI:
    def __init__(self, traps, goal):
        pygame.init()
        self.screen = pygame.display.set_mode((MAZE_SIZE, MAZE_SIZE + OFFSET_Y))
        pygame.display.set_caption("秘境寻宝")
        self.font = pygame.font.SysFont("SimHei", 20)
        self.clock = pygame.time.Clock()
        self.traps = traps
        self.goal = goal

    def draw(self, position, episode, step):
        self.screen.fill(COLOR_BG)

        # 1. 状态信息
        info = self.font.render(f"episode: {episode + 1} | step: {step} | trap: {len(self.traps)}", True, COLOR_TEXT)
        self.screen.blit(info, (10, 15))

        # 2. 网格和元素
        for r in range(GRID_COUNT):
            for c in range(GRID_COUNT):
                rect = (c * CELL_SIZE, r * CELL_SIZE + OFFSET_Y, CELL_SIZE, CELL_SIZE)
                pygame.draw.rect(self.screen, COLOR_LINE, rect, 1)  # 画边框

                # 画陷阱
                if (r, c) in self.traps:
                    pygame.draw.rect(self.screen, COLOR_TRAP, (rect[0] + 4, rect[1] + 4, CELL_SIZE - 8, CELL_SIZE - 8))
                # 画终点
                elif (r, c) == self.goal:
                    pygame.draw.rect(self.screen, COLOR_GOAL, (rect[0] + 4, rect[1] + 4, CELL_SIZE - 8, CELL_SIZE - 8))

        # 3. 画探险家
        if position != 'terminal':
            r, c = eval(position)
            center = (c * CELL_SIZE + CELL_SIZE // 2, r * CELL_SIZE + OFFSET_Y + CELL_SIZE // 2)
            pygame.draw.circle(self.screen, COLOR_AGENT, center, CELL_SIZE // 3)

        pygame.display.flip()
```

```python
import torch

# 将坐标字符串转成PyTorch需要的tensor
def state_to_tensor(state_str):
    r, c = eval(state_str)
    return torch.FloatTensor([r, c])
```

```python
import torch.nn as nn

# 定义一个神经网络，输入的n_states是坐标，输出的n_actions是动作对应的Q值
# 相当于使用一个神经网络来记录某个坐标对应的某个动作的Q值
# 一开始神经网络输出的4个动作的Q值是不对的，但是随着训练，Q值会越来越正确
class Net(nn.Module):
    def __init__(self, n_states, n_actions):
        super(Net, self).__init__()
        self.fc = nn.Sequential(
            nn.Linear(n_states, 64),
            nn.ReLU(),
            nn.Linear(64, 64),
            nn.ReLU(),
            nn.Linear(64, n_actions)
        )

    def forward(self, x):
        return self.fc(x)
```

```python
import random

# 初始化环境，比如陷阱数量

TRAP_COUNT = 3  # 陷阱数量

def init_environment():
    """初始化陷阱和终点坐标"""
    all_positions = [(r, c) for r in range(GRID_COUNT) for c in range(GRID_COUNT)]
    all_positions.remove((0, 0))  # 起点不能是陷阱
    goal = (GRID_COUNT - 1, GRID_COUNT - 1)
    all_positions.remove(goal)  # 终点不能是陷阱

    # 随机选出陷阱
    traps = random.sample(all_positions, min(TRAP_COUNT, len(all_positions)))
    return traps, goal
```

```python
import numpy as np

ACTIONS = ['up', 'down', 'left', 'right']

# 用神经网络获取动作，dqn：deep q network
def choose_action_dqn(state, epsilon, net):
    state_tensor = state_to_tensor(state)

    if np.random.uniform() < epsilon:
        # 将坐标（当前状态）输入给神经网络，得到每个动作的Q值，然后取Q值最大的动作
        actions_value = net(state_tensor)
        action_idx = torch.argmax(actions_value).item()
    else:
        action_idx = np.random.randint(0, len(ACTIONS))

    return ACTIONS[action_idx]
```

```python
# 与环境交互，获取下一个状态和奖励
def get_env_feedback(current_state, action):
    # current_state是字符串"(0, 0)"，用eval函数解析，r_idx表示当前状态的行索引，c_idx表示当前状态的列索引
    r_idx, c_idx = eval(current_state)

    # 根据动作决定下一个位置的坐标
    if action == 'up':
        r_idx = max(0, r_idx - 1)
    elif action == 'down':
        r_idx = min(GRID_COUNT - 1, r_idx + 1)
    elif action == 'left':
        c_idx = max(0, c_idx - 1)
    elif action == 'right':
        c_idx = min(GRID_COUNT - 1, c_idx + 1)

    next_state = (r_idx, c_idx)

    is_terminated = False
    if next_state == goal:
        # 得到宝藏
        reward = 1
        is_terminated = True
    elif next_state in traps:
        # 遇到陷阱
        reward = -1
        is_terminated = True
    else:
        # 普通步骤
        reward = 0
        # reward = -0.01  # 每走一步扣一点分，鼓励走短路

    return str(next_state), reward, is_terminated
```

```python
from torch import optim

traps, goal = init_environment()
gui = MazeGUI(traps, goal)

EPSILON = 0.9  # 贪婪度：90% 的时间选最优动作，10% 随机探索
GAMMA = 0.9  # 未来奖励衰减因子

net = Net(2, len(ACTIONS))   # 估值网络
optimizer = optim.Adam(net.parameters(), lr=0.0001)
loss_func = nn.MSELoss()
BATCH_SIZE = 32

total_steps = 0
for episode in range(500):
    current_state = "(0, 0)"
    is_terminated = False

    episode_loss = 0
    loss_count = 0
    while not is_terminated:

        # 用net选择一个动作
        action = choose_action_dqn(current_state, EPSILON, net)
        action_idx = ACTIONS.index(action)

        # 执行动作
        next_s, reward, is_terminated = get_env_feedback(current_state, action)

        # 记录动作
        memory.append((state_to_tensor(current_state), action_idx, reward, state_to_tensor(next_s), is_terminated))

        if len(memory) > BATCH_SIZE:

            # 从最近的动作中采样一个batch的动作，假如BATCH_SIZE=1，那么拿到的就是历史记录中走过的某一次动作
            # 从b_s状态执行了b_a动作，达到了b_s_新状态，得到了b_r奖励，b_done表示该动作是不是结束动作（获得奖励或遇到陷阱）
            batch = random.sample(memory, BATCH_SIZE)
            b_s, b_a, b_r, b_s_, b_done = zip(*batch)

            # 把上面4个都转成tensor
            b_s = torch.stack(b_s)
            b_a = torch.LongTensor(b_a).view(-1, 1)
            b_r = torch.FloatTensor(b_r).view(-1, 1)
            b_s_ = torch.stack(b_s_)
            b_done = torch.FloatTensor(np.array(b_done).astype(float)).view(-1, 1) # 关键：转换为 0/1

            # 得到当前net中b_s和b_a对应的Q值
            # 输入b_s当前坐标，输出4个动作的Q值，gather负责从4个动作中取出b_a动作的Q值
            q = net(b_s).gather(1, b_a)

            # 得到b_s_，也就是下一个状态对应的动作中最大的Q值，也就是未来奖励
            with torch.no_grad():
                # 输入下一个状态坐标，输出4个动作的Q值，detach()不进行梯度计算，max()取出最大的Q值，也就是未来奖励
                q_next = net(b_s_).detach().max(1)[0].view(-1, 1)
                # b_r是当前状态的奖励，q_next是未来奖励，q_target就是新的Q值
                # 如果遇到终点或者陷阱，没有未来奖励
                q_target = b_r + GAMMA * q_next * (1 - b_done)


            # 神经网络反向传播，优化net，使得输出的Q值越来越接近q_target
            loss = loss_func(q, q_target)
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            # 记录损失
            episode_loss += loss.item()
            loss_count += 1

        # if epoch % 50 == 0: # 减少绘图频率加速训练
        #     gui.draw(current_state, epoch, total_steps)
        #     pygame.time.delay(20)

        current_state = next_s
        total_steps += 1

    avg_loss = episode_loss / max(1, loss_count)

    if episode % 50 == 0:
        print(f"Episode: {episode} | Memory: {len(memory)} | Loss: {avg_loss}")

print("DQN 训练结束！")
```

```python
current_state = "(0, 0)"

while not is_terminated:
```
