```python
import torch


# 生成训练数据
def generate_training_data(sentences, window_size):
    context_words = []  #周围词
    target_words = []  #中心词

    for sentence in sentences:
        indices = [vocabulary[word] for word in sentence]
        for center_pos in range(len(indices)):
            # 确定上下文窗口范围
            start = max(0, center_pos - window_size)
            end = min(len(indices), center_pos + window_size + 1)

            # 收集上下文词（排除中心词本身）
            context = []
            for context_pos in range(start, end):
                if context_pos != center_pos:
                    context.append(indices[context_pos])

            # 至少需要1个上下文词
            if len(context) >= 1:
                context_words.append(context)
                target_words.append(indices[center_pos])

    return context_words, target_words


WINDOW_SIZE = 2
```

```python
from torch import nn
from torch import optim


# 定义CBOW模型
class CBOW(nn.Module):
    def __init__(self, vocab_size, embedding_dim):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        self.linear = nn.Linear(embedding_dim, vocab_size)

    def forward(self, context_words):
        hidden = self.embedding(context_words) # 输入是周围词的下标
        # 取所有周围词对应的向量的平均值，dim=1表示按第1维求平均，也就是把4个单词对应的向量加起来求平均，得到一个(1, 100)的向量
        avg = torch.mean(hidden, dim=1)
        # 最终得到一个(1, vocab_size)的结果
        return self.linear(avg)
```

```python
EMBEDDING_DIM = 100

# 初始化模型、损失函数和优化器
model = CBOW(vocab_size, EMBEDDING_DIM)
criterion = nn.CrossEntropyLoss()
optimizer = optim.SGD(model.parameters(), lr=0.01)
```

```python
# 开始训练
EPOCHS = 10

for epoch in range(EPOCHS):
    for context_batch, target_batch in dataloader:
        # 前向传播
        predict = model(context_batch)

        # 计算损失
        loss = criterion(predict, target_batch)

        # 反向传播
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        print(f"Epoch {epoch + 1}/{EPOCHS}, Loss: {loss:.4f}")
```

## SkipGram和CBOW的对比

SkipGram在训练时，是一个词预测多个词，然后取概率最高的，所以它更能找出中心词和哪个词是更为接近的，也就是更能识别出词和词之间的语义关系。
SkipGram在训练时，是按词对来进行训练的，所以训练起来需要的时间相对更久一点。
SkipGram相当于训练的更精细一点，适合小数据集。

CBOW是多个词预测一个词，训练会更快，CBOW训练的更粗放一点，适合大数据集。
