```java
package com.example.zxds.mps.service.impl;

import cn.hutool.http.HttpRequest;
import cn.hutool.http.HttpResponse;
import cn.hutool.json.JSONArray;
import cn.hutool.json.JSONObject;
import cn.hutool.json.JSONUtil;
import com.example.zxds.common.exception.ZxdsException;
import com.example.zxds.mps.service.MapService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.HashMap;
import java.util.Map;

@Service
@Slf4j
public class MapServiceImpl implements MapService {
    
    //预估里程的API地址
    private String distanceUrl = "https://apis.map.qq.com/ws/distance/v1/matrix/";

    @Value("${tencent.map.key}")
    private String key;

    @Override
    public Map<String, Object> estimateOrderMileageAndMinute(String mode, String startPlaceLatitude, String startPlaceLongitude, String endPlaceLatitude, String endPlaceLongitude) {
        HttpRequest req = new HttpRequest(distanceUrl);
        req.form("mode", mode);
        req.form("from", startPlaceLatitude + "," + startPlaceLongitude);
        req.form("to", endPlaceLatitude + "," + endPlaceLongitude);
        req.form("key", key);
        HttpResponse resp = req.execute();
        JSONObject json = JSONUtil.parseObj(resp.body());
        int status = json.getInt("status");
        String message = json.getStr("message");
        System.out.println(message);
        if (status != 0) {
            log.error(message);
            throw new ZxdsException("预估里程异常：" + message);
        }
        JSONArray rows = json.getJSONObject("result").getJSONArray("rows");
        JSONObject element = rows.get(0, JSONObject.class).getJSONArray("elements").get(0, JSONObject.class);
        int distance = element.getInt("distance");
        // 米转换为公里
        String mileage = new BigDecimal(distance).divide(new BigDecimal(1000)).toString();
        int duration = element.getInt("duration");
        // 秒转换为分钟
        String temp = new BigDecimal(duration).divide(new BigDecimal(60), 0, RoundingMode.CEILING).toString();
        int minute = Integer.parseInt(temp);

        Map<String, Object> map = new HashMap<>() {{
            put("mileage", mileage);
            put("minute", minute);
        }};
        return map;
    }
}
```

```java
private String directionUrl = "https://apis.map.qq.com/ws/direction/v1/driving/";

public Map<String, Object> calculateDriveLine(String startPlaceLatitude, String startPlaceLongitude, String endPlaceLatitude, String endPlaceLongitude) {
    String url = directionUrl + "?from=" + startPlaceLatitude + "," + startPlaceLongitude
            + "&to=" + endPlaceLatitude + "," + endPlaceLongitude
            + "&key=" + key;

    // 2. 使用HttpRequest.get()方法（核心修改）
    HttpRequest req = HttpRequest.get(url);

    HttpResponse resp = req.execute();
    JSONObject json = JSONUtil.parseObj(resp.body());
    System.out.println(json);
    int status = json.getInt("status");
    if (status != 0) {
        throw new ZxdsException("执行异常");
    }
    JSONObject result = json.getJSONObject("result");
    Map<String, Object> map = result.toBean(Map.class);
    return map;
}
```

## 开启实时定位功能

开启实时定位功能，只需要开启一次就行了。怎么保证实时定位开启一次？在根组件 `App.vue`的 `onLaunch`回调函数中开启即可。

onLaunch 是 uni-app 特有的应用生命周期函数，它只存在于 App.vue 这个根组件中。

**执行时机**：只在**整个应用第一次启动时**执行一次。

**找到 **`**<font style="color:#000000;">zxds-customer-wx</font>**`**小程序中的 **`**<font style="color:#000000;">App.vue</font>**`**页面，编写以下代码开启 GPS 实时定位。并且当定位发生变化时自动执行回调函数，代码如下：**

```javascript
onLaunch: function() {
    //开启GPS后台刷新
    wx.startLocationUpdate({
        success(resp) {
            console.log('开启定位成功');
        },
        fail(resp) {
            console.log('开启定位失败');
        }
    });
    // 当GPS定位发生变化时，自动执行回调函数
    // LocationChange是一个事件的名字，当GPS定位发生变化时，自动调用回调函数
    wx.onLocationChange(function(resp) {
        // 拿到最新的维度
        let latitude = resp.latitude;
        // 拿到最新的经度
        let longitude = resp.longitude;
        // 封装为对象，方便将数据传递到工作台页面
        let location = { latitude: latitude, longitude: longitude };
        //触发自定义事件（在工作台页面绑定了 updateLocation事件，以下代码是触发这个事件，通过这种方式将数据传递给工作台页面）
        uni.$emit('updateLocation', location);
    });
},
```

`**<font style="color:rgb(15, 17, 21);background-color:rgb(235, 238, 242);">wx</font>**`** 是微信小程序专用的原生对象，**`**<font style="color:rgb(15, 17, 21);background-color:rgb(235, 238, 242);">uni</font>**`** 是 uni-app 框架提供的跨平台统一对象。**

用 `<font style="color:rgb(15, 17, 21);background-color:rgb(235, 238, 242);">wx</font>` 写代码只能跑在微信里，用 `<font style="color:rgb(15, 17, 21);background-color:rgb(235, 238, 242);">uni</font>` 写同一份代码能编译成微信小程序、App、H5等多家平台。

`<font style="color:rgb(15, 17, 21);">uni</font>`对象对 `<font style="color:rgb(15, 17, 21);">wx</font>`对象进行了二次包装。因此尽量使用 `<font style="color:rgb(15, 17, 21);">uni</font>`，不要使用 `<font style="color:rgb(15, 17, 21);">wx</font>`。

## 工作台上的地图是怎么展示出来的

```html
<!--id 组件的唯一标识符-->
```

## 工作台上的地图位置实现实时更新

`**<font style="color:#000000;">pages/workbench/workbench.vue</font>**`

**怎么实现工作台页面上的地图位置实时更新呢？原理是：**

- 当用户位置发生变化时，`<font style="color:#000000;">App.vue</font>`中的 `<font style="color:#000000;">uni.$emit('updateLocation', location);</font>`就会执行，触发 `<font style="color:#000000;">updateLocation</font>`事件，并将当前最新的 `<font style="color:#000000;">location</font>`数据传递给绑定了 `<font style="color:#000000;">updateLocation</font>`事件的组件。
- 我们接下来的代码就是在工作台 `<font style="color:#000000;">workbench.vue</font>`组件上绑定 `<font style="color:#000000;">updateLocation</font>`事件。
- 工作台页面一旦绑定 `<font style="color:#000000;">updateLocation</font>`事件后，就能实时接收到最新的 `<font style="color:#000000;">location</font>`对象。
- 从 `<font style="color:#000000;">location</font>`对象中取出经纬度，赋值给模型层的变量，地图的位置就实时更新了。

**先来看一下工作台页面 **`**<font style="color:#000000;">pages/workbench/workbench.vue</font>**`**的模型层代码：**

```javascript

            address: '',
            longitude: 0,
            latitude: 0
        },
        // 终点
        to: {
            address: '输入你的目的地',
            longitude: 0,
            latitude: 0
        },
        // 当前的经度
        longitude: 0,
        // 当前的维度
        latitude: 0,
        contentStyle: '',
        windowHeight: 0,
        map: null,

```

**接下来，我们就来实现一下，我们选择在 **`**<font style="color:#000000;">onShow</font>**`**函数中编写代码，因为我们要达到的效果是：用户只要回到工作台页面，就要刷新位置：**

```javascript
onShow: function() {
    let that = this;
    // 想要操作地图的第一件事：获取地图上下文对象
    that.map = uni.createMapContext('map');
    // QQMapWX 是腾讯地图官方为微信小程序和网页提供的 JavaScript SDK
    // 用来在小程序里快速调用地图服务（如定位、搜索、路线规划、地址解析等）
    let qqmapsdk = new QQMapWX({
        key: that.tencent.map.key
    });
    // 绑定 updateLocation 事件。只要事件发生，回调函数执行，拿到最新location，更新经纬度变量值，达到更新地图位置。
    uni.$on('updateLocation', function(location) {
        //console.log(location);
        // 当用户手动选点时，flag不会为空。也就是当用户手动选点时，不走以下的逻辑，因为一旦走以下逻辑，自动定位的信息会将用户手动选点覆盖掉。
              if(that.flag!=null){
                                    return
                          }
        // 获取最新经度维度
        let latitude = location.latitude;
        let longitude = location.longitude;
        // 将经度维度值赋值给变量
        that.latitude = latitude;
        that.longitude = longitude;
        // 将经度维度设置到上车点
        that.from.latitude = latitude;
        that.from.longitude = longitude;
      
        //把坐标解析成地址（逆向解析，根据坐标得出地名）
        qqmapsdk.reverseGeocoder({
            location: {
                latitude: latitude,
                longitude: longitude
            },
            success: function(resp) {
                //console.log(resp);
                that.from.address = resp.result.address;
            },
            fail: function(error) {
                console.log(error);
            }
        });
    });
},
onHide: function() {
    // 当工作台页面隐藏的时候，记得解除事件绑定：意思就是工作台页面既然隐藏起来了，就没必要再浪费手机电量去更新地图位置了。
    uni.$off('updateLocation');
},
```

[https://lbs.qq.com/miniProgram/jsSdk/jsSdkGuide/methodReverseGeocoder](https://lbs.qq.com/miniProgram/jsSdk/jsSdkGuide/methodReverseGeocoder)

核心代码是：

```javascript

  qqmapsdk.reverseGeocoder({

      },

      }
  });
}
```

### 封装计算最简线路的函数

计算**最佳线路、预估里程和时长**的代码我们可以用一个函数给封装起来，将来使用的时候调用起来非常方便。放到 `<font style="color:#000000;">create_order.vue</font>`的 `<font style="color:#000000;">methods:</font>`中。

```javascript
calculateLine(ref) {
    // 调用腾讯地图API的路线规划服务（驾车模式）
    qqmapsdk.direction({
        mode: 'driving', // 交通方式：驾车
        from: { // 起点坐标
            latitude: ref.from.latitude,
            longitude: ref.from.longitude
        },
        to: { // 终点坐标
            latitude: ref.to.latitude,
            longitude: ref.to.longitude
        },
        success: function(resp) {
            // 校验API返回状态
            if (resp.status != 0) {
                uni.showToast({
                    icon: 'error',
                    title: resp.message
                });
                return;
            }
            
            // 获取第一条路线数据
            let route = resp.result.routes[0];
            let distance = route.distance; // 路线距离（米）
            let duration = route.duration; // 路线时间（分钟）
            let polyline = route.polyline; // 路线坐标点（加密字符串）
            
            // 处理距离数据：米转公里并保留一位小数
            ref.distance = Math.ceil((distance / 1000) * 10) / 10;
            ref.duration = duration; // 存储预计时间
            
            // 将加密的折线坐标解析为坐标点数组
            let points = ref.formatPolyline(polyline);
            
            // 设置地图折线（路线）显示配置
            ref.polyline = [
                {
                    points: points, // 路线坐标点数组
                    width: 6,       // 路线宽度
                    color: '#05B473', // 路线颜色
                    arrowLine: true  // 显示方向箭头
                }
            ];
            
            // 设置地图标记点（起点和终点）
            ref.markers = [
                {
                    id: 1, // 起点标记（随便写，保证不重复就行。）
                    latitude: ref.from.latitude,
                    longitude: ref.from.longitude,
                    width: 25, // 图标宽度
                    height: 35, // 图标高度
                    anchor: { // 图标的中心对准坐标
                        x: 0.5,
                        y: 0.5
                    },
                    iconPath: 'https://mapapi.qq.com/web/lbs/javascriptGL/demo/img/start.png' // 起点图标
                },
                {
                    id: 2, // 终点标记
                    latitude: ref.to.latitude,
                    longitude: ref.to.longitude,
                    width: 25,
                    height: 35,
                    anchor: {
                        x: 0.5,
                        y: 0.5
                    },
                    iconPath: 'https://mapapi.qq.com/web/lbs/javascriptGL/demo/img/end.png' // 终点图标
                }
            ];
        }
    });

},
```

在`<font style="color:#000000;background-color:rgba(220, 240, 255, 0.3) !important;">onShow()</font>`回调函数中调用上面封装的函数。为什么不在`<font style="color:#000000;background-color:rgba(220, 240, 255, 0.3) !important;">onLoad()</font>`函数中调用？如果小程序挂起在后台，过了几分钟，又切回到小程序，应该立即重新计算最佳线路，因为有可能有的路段已经开始堵车了。


```java
private final OrderBillDao orderBillDao;

private final StringRedisTemplate stringRedisTemplate;

@Override
@Transactional
public String insertOrder(OrderEntity orderEntity, OrderBillEntity billEntity) {
    //插入订单记录
    int rows = orderDao.insert(orderEntity);
    if (rows == 1) {
        String id = orderDao.searchOrderIdByUUID(orderEntity.getUuid());
        //插入订单费用记录
        billEntity.setOrderId(Long.parseLong(id));
        rows = orderBillDao.insert(billEntity);
        if (rows == 1) {
            //往Redis里面插入缓存，配合Redis事务用于司机抢单，避免多个司机同时抢单成功
            stringRedisTemplate.opsForValue().set("order#" + id, "none");
            stringRedisTemplate.expire("order#" + id, 15, TimeUnit.MINUTES);   //缓存15分钟
            return id;
        } else {
            throw new ZxdsException("保存新订单费用失败");
        }
    } else {
        throw new ZxdsException("保存新订单失败");
    }
}
```

## Redis 的 Geo 功能

创建订单成功后需要推送订单信息给附近的司机。假设司机端的小程序实时把自己的GPS定位上传，然后定位信息缓存到Redis里面。咱们怎么能利用Redis计算出，上车点方圆几公里的司机都有谁呢？这就需要使用Redis的Geo功能。

**进入 Redids docker 容器内部：**

```shell
docker exec -it redis /bin/bash
redis-cli
AUTH abc123456
```

Redis的Geo主要用于存储地理位置信息，并对存储的信息进行操作，该功能在 Redis 3.2 版本新增。 下面我们用`<font style="color:#000000;background-color:rgba(220, 240, 255, 0.3) !important;">GEOADD</font>`命令向Redis里面添加几个景点的定位。

```shell
GEOADD china 116.403963 39.915119 tiananmen 116.417876 39.915411 wangfujing 116.404354 39.904748 qianmen
```

然后我们`<font style="color:#000000;background-color:rgba(220, 240, 255, 0.3) !important;">GEORADIUS</font>`命令查询距离某个定位点1公里范围以内的景点有哪些。

```shell
GEORADIUS china 116.4000 39.9000 1 km WITHDIST
```

![27441359-ce92-4667-8895-1463ab593f45.png](images/RpoxbzfnloKaWExkqzOcIch2nKh.png)

既然Redis的GEO命令可以帮我们提取出某个坐标点指定距离以内的景点，如果Redis里面缓存的是司机的定位信息，那么我们用代驾单的起点坐标来查询附近几公里以内的司机，是不是也可以？而且Redis的Geo计算是在内存中完成的，比MySQL的Geo计算快了上千倍。

有个需要特别注意的问题就是Geo中的定位我们没办法设置超时时间，所以我们想要知道哪些司机上线中，就得额外创建带有超时的缓存。

我的设想是缓存司机信息用的Key是`<font style="color:#000000;background-color:rgba(220, 240, 255, 0.3) !important;">driver_online#driverId</font>`，对应的Value是`**<font style="color:#000000;background-color:rgba(220, 240, 255, 0.3) !important;">接单距离#订单里程范围#定向接单的坐标</font>**`，超时时间为1分钟。

当系统接到订单之后，到Redis上面根据driverId查找缓存，找到了就是在线，找不到就是不在线。

# 位置微服务缓存司机实时定位

我们弄明白了Redis的GEO技术。咱们想要让Redis计算代驾起点周围几公里以内的司机，首先我们要把司机的定位信息缓存到GEO里面，然后还要额外创建司机上线缓存。（不能在 GEO 缓存中缓存司机在线状态，因为设置 GEO 缓存有效期后，GEO 缓存失效的话，所有司机的定位信息会被全部删除，因此要单独再维护一个司机在线的 Redis 缓存。）

```java
package com.example.zxds.mps.service.impl;

import cn.hutool.core.convert.Convert;
import cn.hutool.core.map.MapUtil;
import com.example.zxds.mps.service.DriverLocationService;
import lombok.RequiredArgsConstructor;
import org.springframework.data.geo.Point;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

import java.util.Map;
import java.util.concurrent.TimeUnit;

@Service
@RequiredArgsConstructor
public class DriverLocationServiceImpl implements DriverLocationService {

    private final RedisTemplate redisTemplate;

    @Override
    public void updateLocationCache(Map<String, Object> param) {
        long driverId = MapUtil.getLong(param, "driverId");
        String latitude = MapUtil.getStr(param, "latitude");
        String longitude = MapUtil.getStr(param, "longitude");

        //接单范围
        int rangeDistance = MapUtil.getInt(param, "rangeDistance");
        //订单里程范围
        int orderDistance = MapUtil.getInt(param, "orderDistance");
        //封装成Point对象才能缓存到Redis里面
        Point point = new Point(Convert.toDouble(longitude), Convert.toDouble(latitude));
        /*
         * 把司机实时定位缓存到Redis里面，便于Geo定位计算
         * Geo是集合形式，如果设置过期时间，所有司机的定位缓存就全都失效了
         * 正确做法是司机上线后，更新GEO中的缓存定位
         */
        redisTemplate.opsForGeo().add("driver_location", point, driverId + "");

        //定向接单地址的经度
        String orientateLongitude = null;
        if (param.get("orientateLongitude") != null) {
            orientateLongitude = MapUtil.getStr(param, "orientateLongitude");
        }
        //定向接单地址的纬度
        String orientateLatitude = null;
        if (param.get("orientateLatitude") != null) {
            orientateLatitude = MapUtil.getStr(param, "orientateLatitude");
        }
        //定向接单经纬度的字符串
        String orientation = "none";
        if (orientateLongitude != null && orientateLatitude != null) {
            orientation = orientateLatitude + "," + orientateLongitude;
        }

        /*
         * 为了解决判断哪些司机在线，我们还要单独弄一个上线缓存
         * 缓存司机的接单设置（定向接单、接单范围、订单总里程），便于系统判断该司机是否符合接单条件
         */
        String temp = rangeDistance + "#" + orderDistance + "#" + orientation;
        redisTemplate.opsForValue().set("driver_online#" + driverId, temp, 60, TimeUnit.SECONDS);
    }

    @Override
    public void removeLocationCache(long driverId) {
        //删除司机定位缓存
        redisTemplate.opsForGeo().remove("driver_location", driverId + "");
        //删除司机上线缓存
        redisTemplate.delete("driver_online#" + driverId);
    }
}
```

```java
package com.example.zxds.mps.service.impl;

import cn.hutool.core.convert.Convert;
import cn.hutool.core.map.MapUtil;
import com.example.zxds.mps.service.DriverLocationService;
import com.example.zxds.mps.util.CoordinateTransform;
import lombok.RequiredArgsConstructor;
import org.gavaghan.geodesy.Ellipsoid;
import org.gavaghan.geodesy.GeodeticCalculator;
import org.gavaghan.geodesy.GeodeticCurve;
import org.gavaghan.geodesy.GlobalCoordinates;
import org.springframework.data.geo.*;
import org.springframework.data.redis.connection.RedisGeoCommands;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

import java.util.*;

import java.util.concurrent.TimeUnit;

@Service
@RequiredArgsConstructor
public class DriverLocationServiceImpl implements DriverLocationService {

    //........

    @Override
    public List<Map<String, Object>> searchBefittingDriverAboutOrder(double startPlaceLatitude, double startPlaceLongitude, double endPlaceLatitude, double endPlaceLongitude, double mileage) {
        //搜索订单起始点5公里以内的司机
        Point point = new Point(startPlaceLongitude, startPlaceLatitude);
        //设置GEO距离单位为千米
        Metric metric = RedisGeoCommands.DistanceUnit.KILOMETERS;
        Distance distance = new Distance(5, metric);
        Circle circle = new Circle(point, distance);

        //创建GEO参数
        RedisGeoCommands.GeoRadiusCommandArgs args = RedisGeoCommands.GeoRadiusCommandArgs.newGeoRadiusArgs()
                .includeDistance() //让结果中包含距离
                .includeCoordinates() //让结果中包含坐标
                .sortAscending(); //让结果升序排列

        //执行GEO计算，获得查询结果
        GeoResults<RedisGeoCommands.GeoLocation<String>> radius = redisTemplate.opsForGeo().radius("driver_location", circle, args);

        //需要通知的司机列表
        List<Map<String, Object>> list = new ArrayList<>();

        if (radius != null) {
            // redis的GEO的计算结果是可迭代的，获取迭代器
            Iterator<GeoResult<RedisGeoCommands.GeoLocation<String>>> iterator = radius.iterator();
            // 迭代每一个结果
            while (iterator.hasNext()) {
                GeoResult<RedisGeoCommands.GeoLocation<String>> result = iterator.next();
                // 获取内容
                RedisGeoCommands.GeoLocation<String> content = result.getContent();
                // 从内容中获取司机id
                String driverId = content.getName();
                // 距离中心点的距离
                double dist = result.getDistance().getValue();

                //排除 不在线的司机
                if (!redisTemplate.hasKey("driver_online#" + driverId)) {
                    continue;
                }

                //查找该司机的在线缓存
                Object obj = redisTemplate.opsForValue().get("driver_online#" + driverId);
                //如果查找的那一刻，缓存超时被置空，那么就忽略该司机
                if (obj == null) {
                    continue;
                }

                String value = obj.toString();
                String[] temp = value.split("#");
                int rangeDistance = Integer.parseInt(temp[0]);
                int orderDistance = Integer.parseInt(temp[1]);
                String orientation = temp[2];

                //判断是否符合接单范围
                boolean flag1 = dist <= rangeDistance;

                //判断订单里程是否符合
                boolean flag2 = false;
                if (orderDistance == 0) {
                    flag2 = true;
                } else if (orderDistance == 5 && mileage > 0 && mileage <= 5) {
                    flag2 = true;
                } else if (orderDistance == 10 && mileage > 5 && mileage <= 10) {
                    flag2 = true;
                } else if (orderDistance == 15 && mileage > 10 && mileage <= 15) {
                    flag2 = true;
                } else if (orderDistance == 30 && mileage > 15 && mileage <= 30) {
                    flag2 = true;
                }

                //判断定向接单是否符合
                boolean flag3 = false;
                if (!orientation.equals("none")) {
                    double orientationLatitude = Double.parseDouble(orientation.split(",")[0]);
                    double orientationLongitude = Double.parseDouble(orientation.split(",")[1]);
                    //把定向点的火星坐标转换成GPS坐标
                    double[] location = CoordinateTransform.transformGCJ02ToWGS84(orientationLongitude, orientationLatitude);
                    GlobalCoordinates point1 = new GlobalCoordinates(location[1], location[0]);
                    //把订单终点的火星坐标转换成GPS坐标
                    location = CoordinateTransform.transformGCJ02ToWGS84(endPlaceLongitude, endPlaceLatitude);
                    GlobalCoordinates point2 = new GlobalCoordinates(location[1], location[0]);
                    //这里不需要Redis的GEO计算，直接用封装函数计算两个GPS坐标之间的距离
                    GeodeticCurve geoCurve = new GeodeticCalculator().calculateGeodeticCurve(Ellipsoid.WGS84, point1, point2);

                    //如果定向点距离订单终点距离在3公里以内，说明这个订单和司机定向点是顺路的
                    if (geoCurve.getEllipsoidalDistance() <= 3000) {
                        flag3 = true;
                    }

                } else {
                    flag3 = true;
                }

                //匹配接单条件
                if (flag1 && flag2 && flag3) {
                    // 返回距离是因为要在司机端小程序播报订单的距离。（例如：距您3.2公里有一个订单....）
                    Map<String, Object> map = Map.of("driverId", driverId, "distance", dist);
                    list.add(map); //把该司机添加到需要通知的列表中
                }
            }
        }
        return list;
    }
}
```

**火星坐标（GCJ-02）是中国特有的地理坐标系，是在真实GPS坐标（WGS-84）基础上经过人为偏移加密后的坐标，主要用于国内地图服务。**

**GPS坐标（WGS-84）是国际标准**，全球通用。

如果你可以确定整个系统中所有地图都使用火星坐标系，可以不做转换。

# 项目中使用 RabbitMQ

## 为什么选择 RabbitMQ 而非 Kafka

**业务场景匹配度**

```plaintext
代驾系统核心流程：
用户下单 → 系统派单 → 司机接单 → 开始服务 → 完成结算

特点：
- 消息量适中（非海量数据）
- 需要精确路由和可靠投递
- 每条消息都很重要，不能丢失
- 有明确的消费者（特定司机）
```

| **考量维度** | **代驾系统需求** | **RabbitMQ 优势** | **Kafka 劣势** |
| --- | --- | --- | --- |
| **消息路由** | 需要精确推送给特定司机 | Exchange + Routing Key 精确路由 | Topic + Partition，消费者需过滤 |
| **消息确认** | 必须确保司机收到订单 | 完善的ACK机制，可重试 | At-least-once语义，可能重复 |
| **消息优先级** | 紧急订单优先处理 | 支持消息优先级 | 不支持（FIFO顺序） |
| **消息TTL** | 订单超时自动取消 | 原生支持消息/队列TTL | 仅支持整体保留时间 |
| **死信队列** | 处理失败订单 | 内置死信交换器 | 需要额外实现 |
| **实时性** | 秒级响应 | 毫秒级延迟 | 吞吐优先，延迟较高 |

## 我们主要使用哪种模式

我们主要使用发布订阅模式，并且交换机类型使用 `<font style="color:#000000;">Direct</font>`类型：**每一个司机都有自己的消息队列。每个消息队列的路由键采用 **`**<font style="color:#000000;">driverId</font>**`。

# 消息微服务收发新订单消息

## 发送消息

当用户下单时，给符合条件的司机发送消息，这里我们采用异步方式发送，也就是说**立即让乘客收到下单成功的提示信息**，乘客不用等到给司机发完消息之后才收到下单成功的消息。当然，我们这里编写的时候，同步和异步方式都可以提供。至于说将来程序员使用哪种方式发送消息，他可以自己决定。

### 同步发送

在`<font style="color:#000000;background-color:rgba(220, 240, 255, 0.3) !important;">com.example.zxds.snm.task</font>`包中创建`<font style="color:#000000;background-color:rgba(220, 240, 255, 0.3) !important;">NewOrderMassageTask.java</font>`类，在里面定义同步发送消息的代码。

```java
package com.example.zxds.snm.task;

import com.example.zxds.snm.entity.NewOrderMessage;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.rabbitmq.client.GetResponse;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.amqp.core.AmqpAdmin;
import org.springframework.amqp.rabbit.core.RabbitTemplate;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

@Component
@Slf4j
@RequiredArgsConstructor
public class NewOrderMessageTask {

    // Spring Boot 自动配置的 RabbitTemplate 实例，用于发送和接收消息
    private final RabbitTemplate rabbitTemplate;
    // Spring AMQP 管理接口，用于管理交换机、队列等
    private final AmqpAdmin amqpAdmin;
    // 交换机名称，所有新订单消息都通过此交换机路由
    private static final String EXCHANGE_NAME = "new_order_private";
    // 消息过期时间：1分钟（单位：毫秒）
    private static final int TTL = 60 * 1000;
    // JSON 转换器，用于消息体的序列化和反序列化（复用同一个实例避免性能开销）
    private static final ObjectMapper objectMapper = new ObjectMapper();

    /**
     * 同步发送新订单消息（注意：此方法不会创建队列，不会绑定路由键。测试时，先执行接收消息的方法。）
     * @param list 新订单消息列表
     */
    public void sendNewOrderMessage(List<NewOrderMessage> list) {
        for (NewOrderMessage message : list) {
            // convertAndSend：将消息对象自动转换为 JSON 后发送
            // 参数1：交换机名称
            // 参数2：路由键（使用 userId，实现消息路由到对应用户的队列）
            // 参数3：消息对象
            // 参数4：消息处理器，用于设置消息属性（这里设置过期时间）
            rabbitTemplate.convertAndSend(EXCHANGE_NAME, message.getUserId(), message, msg -> {
                msg.getMessageProperties().setExpiration(String.valueOf(TTL));
                return msg;
            });
            log.info("发送成功: {}", message.getOrderId());
        }
    }
    
}
```

## 接收消息

对于司机端接收消息，我采用的是司机端小程序采用 HTTP 请求每隔 3 秒进行一次轮询，因此接收消息不能采用异步方式，你想一下如果采用异步方式的话，主线程已经返回了 R 对象，结果 R 对象中的消息还没有回来。此时 HTTP 的连接都已经断开了。所以接收消息只能采用同步方式。

**另外我们的消息为什么没有采用 WebSocket 推送的方式？而是采用小程序轮询的方式？这是很重要的，对于 Web 系统来说推荐使用 WebSocket 进行推送，但对于小程序来说，信号极不稳定，不推荐使用 WebSocket。**

在 `<font style="color:#000000;">NewOrderMassageTask</font>`类中继续添加接收消息的方法，代码如下：

```java
/**
 * 同步接收新订单消息（轮询方式）
 * @param userId 司机用户ID
 * @return 新订单消息列表（按时间倒序，新消息在前）
 */
public List<NewOrderMessage> receiveNewOrderMessage(long userId) {
    String queueName = "queue_" + userId;
    List<NewOrderMessage> list = new ArrayList<>();

    // execute：在 RabbitMQ Channel 中执行操作，自动管理连接和通道
    return rabbitTemplate.execute(channel -> {
        // 声明队列（如果不存在则自动创建）
        // 参数：队列名，持久化，不独占，不自动删除，无额外参数
        channel.queueDeclare(queueName, true, false, false, null);
        // 将队列绑定到交换机，使用 userId 作为路由键
        channel.queueBind(queueName, EXCHANGE_NAME, String.valueOf(userId));

        GetResponse response;
        // basicGet：非阻塞获取消息，如果没有消息立即返回 null
        while ((response = channel.basicGet(queueName, false)) != null) {
            // 从消息体反序列化为 NewOrderMessage 对象
            NewOrderMessage msg = objectMapper.readValue(response.getBody(), NewOrderMessage.class);
            list.add(msg);
            // 手动确认消息（参数2：false 表示只确认当前消息）
            channel.basicAck(response.getEnvelope().getDeliveryTag(), false);
        }

        // 翻转列表，使新消息排在前面
        Collections.reverse(list);
        return list;
    });

}
```
